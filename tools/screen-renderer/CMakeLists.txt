# Desktop renderer for the air monitor's screen: builds the firmware's own UI
# code and LovyanGFX for the desktop, with stand-ins for the ESP32 parts
# (shim/). Works with Visual Studio's compiler on Windows (build.ps1 sets it
# up) and with GCC or Clang on Linux (build.sh).

cmake_minimum_required(VERSION 3.20)
project(screen_renderer LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(FIRMWARE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../.." CACHE PATH "The firmware")
if(NOT EXISTS "${FIRMWARE_DIR}/src/ui/UiDraw.cpp")
  message(FATAL_ERROR "No firmware in ${FIRMWARE_DIR}; set FIRMWARE_DIR.")
endif()

# LovyanGFX: the version in the firmware's platformio.ini, downloaded once
# (about 50 MB) and checked against its checksum. Or a copy already on disk,
# e.g. the one PlatformIO downloaded: -DLOVYANGFX_DIR=.../libdeps/<env>/LovyanGFX
set(LOVYANGFX_DIR "" CACHE PATH "A LovyanGFX 1.2.33 already on disk (optional)")
if(NOT LOVYANGFX_DIR)
  include(FetchContent)
  FetchContent_Declare(lovyangfx_source
    URL https://github.com/lovyan03/LovyanGFX/archive/refs/tags/1.2.33.tar.gz
    URL_HASH SHA256=17f70bdd5ec21965be7de0a3622ead9e9ce6a73590ecf8ef58ece057b5e87070
    DOWNLOAD_EXTRACT_TIMESTAMP TRUE
    SOURCE_SUBDIR none)  # sources only, without LovyanGFX's own CMake setup
  FetchContent_MakeAvailable(lovyangfx_source)
  set(LOVYANGFX_DIR "${lovyangfx_source_SOURCE_DIR}")
endif()
if(NOT EXISTS "${LOVYANGFX_DIR}/src/LovyanGFX.hpp")
  message(FATAL_ERROR "No LovyanGFX in ${LOVYANGFX_DIR}.")
endif()

# The firmware's version, from platformio.ini, for the Device page.
file(STRINGS "${FIRMWARE_DIR}/platformio.ini" version_line REGEX "FW_VERSION=")
string(REGEX MATCH "FW_VERSION=[^0-9]*([0-9][0-9A-Za-z.+-]*)" _ "${version_line}")
set(FW_VERSION "${CMAKE_MATCH_1}")
if(NOT FW_VERSION)
  set(FW_VERSION "dev")
endif()

if(MSVC)
  add_compile_options(/utf-8 /Zc:__cplusplus /bigobj)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS)
endif()

# LovyanGFX, with its desktop (SDL) back end selected but not built: see
# shim/SDL.h. Only the parts the UI draws with are needed.
set(lgfx "${LOVYANGFX_DIR}/src/lgfx")

# The Chinese, Japanese and Korean fonts are ~120 MB of C source the UI never
# uses; link one-byte placeholders for their tables instead.
set(font_placeholders "${CMAKE_CURRENT_BINARY_DIR}/cjk_font_placeholders.c")
set(content "/* Generated by CMakeLists.txt: placeholders for LovyanGFX's CJK font tables. */\n")
string(APPEND content "#include <stdint.h>\n")
file(GLOB cjk_headers "${lgfx}/Fonts/efont/*.h" "${lgfx}/Fonts/IPA/*.h")
foreach(header IN LISTS cjk_headers)
  file(STRINGS "${header}" declarations REGEX "extern const uint8_t [A-Za-z0-9_]+\\[\\];")
  foreach(declaration IN LISTS declarations)
    string(REGEX REPLACE ".*extern const uint8_t ([A-Za-z0-9_]+)\\[\\];.*" "\\1" name "${declaration}")
    string(APPEND content "const uint8_t ${name}[1] = {0};\n")
  endforeach()
endforeach()
file(CONFIGURE OUTPUT "${font_placeholders}" CONTENT "${content}" @ONLY)

add_library(lovyangfx STATIC
  "${lgfx}/v1/LGFXBase.cpp"
  "${lgfx}/v1/lgfx_fonts.cpp"
  "${lgfx}/v1/lgfx_v1.cpp"
  lgfx/panels.cpp  # instead of lgfx_v1_panel.cpp
  "${lgfx}/utility/lgfx_miniz.c"
  "${lgfx}/utility/lgfx_pngle.c"
  "${lgfx}/utility/lgfx_qoi.c"
  "${lgfx}/utility/lgfx_qrcode.c"
  "${lgfx}/utility/lgfx_tjpgd.c"
  "${font_placeholders}")
target_include_directories(lovyangfx PUBLIC shim "${LOVYANGFX_DIR}/src")
target_compile_definitions(lovyangfx PUBLIC LGFX_SDL)
# Third-party code: no warnings.
if(MSVC)
  target_compile_options(lovyangfx PRIVATE /w "/FI${CMAKE_CURRENT_SOURCE_DIR}/shim/msvc_compat.h")
else()
  target_compile_options(lovyangfx PRIVATE -w)
endif()

add_executable(screen-renderer
  src/main.cpp
  src/Fakes.cpp
  src/Platform.cpp
  src/Png.cpp
  src/Simulator.cpp
  src/World.cpp
  # The firmware's own code.
  "${FIRMWARE_DIR}/src/ui/Ui.cpp"
  "${FIRMWARE_DIR}/src/ui/UiDraw.cpp"
  "${FIRMWARE_DIR}/src/ui/Text.cpp"
  "${FIRMWARE_DIR}/lib/aqcore/src/AirQuality.cpp"
  "${FIRMWARE_DIR}/lib/aqcore/src/ButtonLogic.cpp"
  "${FIRMWARE_DIR}/lib/aqcore/src/Comfort.cpp"
  "${FIRMWARE_DIR}/lib/aqcore/src/History.cpp"
  "${FIRMWARE_DIR}/lib/powercore/src/BatteryWatch.cpp")
target_include_directories(screen-renderer PRIVATE
  src "${FIRMWARE_DIR}/src" "${FIRMWARE_DIR}/lib/aqcore/src" "${FIRMWARE_DIR}/lib/powercore/src")
target_compile_definitions(screen-renderer PRIVATE FW_VERSION="${FW_VERSION}")
if(MSVC)
  # Ui.cpp compares digitalRead()'s int with a bool, as Arduino code does.
  target_compile_options(screen-renderer PRIVATE /wd4805)
endif()
target_link_libraries(screen-renderer PRIVATE lovyangfx)
